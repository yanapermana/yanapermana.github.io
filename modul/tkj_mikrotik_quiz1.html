Menyimpan Nama, Kelas, dan Skor ke Google Spreadsheet "Mikrotik Quiz 1"...

1. Buat Google Spreadsheet baru dengan nama file Mikrotik Quiz 1.
2. Buka Extensions → Apps Script, tempel kode berikut, lalu klik Deploy → New deployment → Web app (Akses: Anyone), dan masukkan URL Web App ke variabel GOOGLE_SCRIPT_URL pada kode HTML ini: 

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheets()[0];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Waktu", "Nama", "Kelas", "Skor", "Spreadsheet"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date(),
    data.nama,
    data.kelas,
    data.skor,
    data.spreadsheet || "Mikrotik Quiz 1"
  ]);
  return ContentService
    .createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
